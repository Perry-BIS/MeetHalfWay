\subsection{Multi-participant Fairness}

% The math scaffold below is ready to use — write the surrounding explanation as prose.

% Shared feasible region over participants P = {1..n} with isochrones R_i:
%   S = \bigcap_{i=1}^{n} R_i
% Empty-intersection handling: report "too far apart" or relax to a union/threshold mode (mark as less fair).
% Per-candidate p travel times T(l_i, p); fairness measured on the SPREAD across participants, e.g.
%   gap(p) = max_i T(l_i,p) - min_i T(l_i,p)
% Penalty grows faster than linearly so very unbalanced options drop quickly (e.g. exp form).
% Mention natural-barrier subtraction (rivers/highways) from S; radius/time-tolerance fallback when
% exact isochrones are unavailable.
\textit{[TODO — Multi-participant Fairness, ~280--360 words. Explain the formulas below.]}
\begin{equation}
S \;=\; \bigcap_{i=1}^{n} R_i ,
\qquad
\mathrm{gap}(p) \;=\; \max_{i} T(l_i, p) \;-\; \min_{i} T(l_i, p).
\label{eq:fairness}
\end{equation}
